// Componente de Gráfico de Linha Moderno com tooltip e linha suave
const GraficoLinhaModerno = ({ periodo, setPeriodo, indicePeriodo, setIndicePeriodo, periodos, labels, series }: any) => {
  const [tooltip, setTooltip] = useState<{ x: number; y: number; label: string; values: any; index: number } | null>(null)
  const svgRef = useRef<SVGSVGElement>(null)
  
  const maxValue = Math.max(...series.flatMap((s: any) => s.data))
  const height = 320
  const width = 900
  const padding = { top: 25, right: 35, bottom: 35, left: 45 }
  const chartHeight = height - padding.top - padding.bottom
  const chartWidth = width - padding.left - padding.right
  
  const getY = (value: number) => padding.top + chartHeight - (value / maxValue) * chartHeight
  const getXPos = (index: number) => padding.left + (index / (labels.length - 1)) * chartWidth
  
  // Função para gerar linha suave (curva bezier)
  const getSmoothPath = (dataPoints: number[]) => {
    if (dataPoints.length < 2) return ''
    let path = `M ${getXPos(0)} ${getY(dataPoints[0])}`
    for (let i = 1; i < dataPoints.length; i++) {
      const prevX = getXPos(i - 1)
      const prevY = getY(dataPoints[i - 1])
      const currX = getXPos(i)
      const currY = getY(dataPoints[i])
      const cp1x = prevX + (currX - prevX) * 0.3
      const cp1y = prevY
      const cp2x = currX - (currX - prevX) * 0.3
      const cp2y = currY
      path += ` C ${cp1x} ${cp1y}, ${cp2x} ${cp2y}, ${currX} ${currY}`
    }
    return path
  }
  
  const handleMouseMove = (e: React.MouseEvent<SVGSVGElement>, idx: number, values: any) => {
    const rect = e.currentTarget.getBoundingClientRect()
    const mouseX = e.clientX - rect.left
    const mouseY = e.clientY - rect.top
    setTooltip({
      x: mouseX,
      y: mouseY - 40,
      label: labels[idx],
      values: values,
      index: idx
    })
  }
  
  const handleMouseLeave = () => {
    setTooltip(null)
  }
  
  const handlePeriodoAnterior = () => {
    if (indicePeriodo < periodos.length - 1) setIndicePeriodo(indicePeriodo + 1)
  }
  
  const handlePeriodoProximo = () => {
    if (indicePeriodo > 0) setIndicePeriodo(indicePeriodo - 1)
  }

  // Obter nome do período atual
  const getPeriodoNome = () => {
    if (periodo === 'semana') return 'Semanal (dados diários)'
    if (periodo === 'mes') return 'Mensal (dados semanais)'
    return 'Anual (dados mensais)'
  }

  return (
    <div className="bg-white rounded-xl shadow-md p-5">
      <div className="flex flex-wrap justify-between items-center gap-3 mb-4">
        <h3 className="font-semibold text-gray-800 text-base">Crescimento de Usuários</h3>
        <div className="flex items-center gap-2">
          <div className="flex items-center gap-1 bg-gray-100 rounded-lg p-0.5">
            <button
              onClick={handlePeriodoAnterior}
              className="p-1.5 hover:bg-white rounded-md transition text-gray-600"
              title="Período anterior"
            >
              <ChevronLeft size={16} />
            </button>
            <div className="relative">
              <select
                value={periodo}
                onChange={(e) => { setPeriodo(e.target.value); setIndicePeriodo(0) }}
                className="px-3 py-1.5 border-0 bg-transparent text-sm text-gray-700 font-medium appearance-none cursor-pointer pr-6 focus:outline-none"
              >
                <option value="semana">📊 Semanal</option>
                <option value="mes">📈 Mensal</option>
                <option value="ano">📉 Anual</option>
              </select>
              <ChevronDown size={14} className="absolute right-1 top-1/2 transform -translate-y-1/2 text-gray-400 pointer-events-none" />
            </div>
            <button
              onClick={handlePeriodoProximo}
              disabled={indicePeriodo === 0}
              className={`p-1.5 rounded-md transition ${indicePeriodo === 0 ? 'text-gray-300 cursor-not-allowed' : 'hover:bg-white text-gray-600'}`}
              title="Próximo período"
            >
              <ChevronRight size={16} />
            </button>
          </div>
        </div>
      </div>

      <p className="text-xs text-gray-500 mb-4 text-center">
        {periodos[indicePeriodo]?.nome || 'Período atual'} • {getPeriodoNome()}
      </p>

      <div className="relative overflow-x-auto">
        <svg 
          ref={svgRef}
          className="w-full" 
          viewBox={`0 0 ${width} ${height}`} 
          preserveAspectRatio="xMidYMid meet"
          style={{ cursor: 'crosshair' }}
        >
          {/* Grid lines suaves */}
          {[0, 20, 40, 60, 80, 100].map((p, i) => {
            const y = getY(maxValue * p / 100)
            return (
              <g key={i} className="opacity-30">
                <line x1={padding.left} y1={y} x2={width - padding.right} y2={y} stroke="#D1D5DB" strokeWidth="0.8" strokeDasharray="3 3" />
                <text x={padding.left - 8} y={y + 3} fontSize="10" fill="#9CA3AF" textAnchor="end">{Math.round(maxValue * p / 100)}</text>
              </g>
            )
          })}
          
          {/* Área de preenchimento suave */}
          {series.map((serie: any, idx: number) => (
            <path
              key={`area-${idx}`}
              d={`${getSmoothPath(serie.data)} L ${getXPos(labels.length - 1)} ${getY(0)} L ${getXPos(0)} ${getY(0)} Z`}
              fill={serie.color}
              fillOpacity="0.06"
              className="transition-all duration-500"
            />
          ))}
          
          {/* Linhas suaves do gráfico */}
          {series.map((serie: any, idx: number) => (
            <path
              key={`line-${idx}`}
              d={getSmoothPath(serie.data)}
              fill="none"
              stroke={serie.color}
              strokeWidth="2.5"
              strokeLinecap="round"
              strokeLinejoin="round"
              className="transition-all duration-500"
            />
          ))}
          
          {/* Pontos interativos com tooltip */}
          {labels.map((label: string, i: number) => {
            const valuesAtPoint: any = {}
            series.forEach((serie: any) => {
              valuesAtPoint[serie.name] = serie.data[i]
            })
            const allPoints = series.flatMap(s => s.data[i])
            const maxPoint = Math.max(...allPoints)
            
            return (
              <g 
                key={`point-group-${i}`}
                onMouseMove={(e) => handleMouseMove(e, i, valuesAtPoint)}
                onMouseLeave={handleMouseLeave}
                className="cursor-pointer"
              >
                {series.map((serie: any, sIdx: number) => {
                  const isHighest = serie.data[i] === maxPoint
                  return (
                    <circle
                      key={`point-${sIdx}-${i}`}
                      cx={getXPos(i)}
                      cy={getY(serie.data[i])}
                      r={isHighest ? 5 : 3.5}
                      fill={serie.color}
                      stroke="white"
                      strokeWidth="2"
                      className="transition-all duration-300 hover:r-6"
                    />
                  )
                })}
              </g>
            )
          })}
          
          {/* Labels do eixo X */}
          {labels.map((label: string, i: number) => (
            <text
              key={`label-${i}`}
              x={getXPos(i)}
              y={height - padding.bottom + 15}
              fontSize="10"
              fill="#6B7280"
              textAnchor="middle"
              className="font-medium"
            >
              {label}
            </text>
          ))}
        </svg>
        
        {/* Tooltip flutuante */}
        {tooltip && (
          <div
            className="fixed bg-gray-900 text-white rounded-lg shadow-xl p-3 z-50 pointer-events-none"
            style={{
              left: tooltip.x + 15,
              top: tooltip.y - 10,
              transform: 'translate(-50%, -100%)',
              minWidth: '200px'
            }}
          >
            <div className="text-xs font-semibold text-gray-300 border-b border-gray-700 pb-1.5 mb-1.5">
              {tooltip.label}
            </div>
            {Object.entries(tooltip.values).map(([name, value]: [string, any]) => {
              let color = '#3B82F6'
              if (name === 'Empresas') color = '#10B981'
              else if (name === 'Ativos') color = '#F59E0B'
              else if (name === 'Academia') color = '#8B5CF6'
              return (
                <div key={name} className="flex items-center justify-between gap-6 text-xs py-1">
                  <div className="flex items-center gap-2">
                    <div className="w-2.5 h-2.5 rounded-full" style={{ backgroundColor: color }} />
                    <span>{name}</span>
                  </div>
                  <span className="font-mono font-semibold">{value.toLocaleString()}</span>
                </div>
              )
            })}
          </div>
        )}
      </div>

      {/* Legenda com indicador de linha suave */}
      <div className="flex flex-wrap justify-center gap-6 mt-6 pt-4 border-t border-gray-100">
        {series.map((serie: any, idx: number) => {
          const ultimoValor = serie.data[serie.data.length - 1]
          const primeiroValor = serie.data[0]
          const variacao = ultimoValor - primeiroValor
          return (
            <div key={idx} className="flex items-center gap-2 group cursor-pointer hover:scale-105 transition-transform">
              <div className="w-3 h-3 rounded-full" style={{ backgroundColor: serie.color }} />
              <span className="text-sm text-gray-600 group-hover:text-gray-800">{serie.name}</span>
              <span className={`text-xs font-semibold ${variacao >= 0 ? 'text-green-600' : 'text-red-600'}`}>
                {variacao >= 0 ? '+' : ''}{variacao}
              </span>
            </div>
          )
        })}
      </div>
    </div>
  )
}

// Componente de Gráfico de Pizza
const GraficoPizza = ({ data, title }: any) => {
  const total = data.reduce((sum: number, item: any) => sum + item.value, 0)
  const DEFAULT_COLORS = ['#3B82F6', '#10B981', '#F59E0B', '#EF4444', '#8B5CF6', '#EC4899']
  let cumulativeAngle = 0
  return (
    <div className="bg-white rounded-xl shadow-md p-5">
      <h3 className="font-semibold text-gray-700 mb-4">{title}</h3>
      <div className="flex flex-col items-center">
        <div className="relative w-40 h-40">
          <svg viewBox="0 0 100 100" className="w-full h-full">
            {data.map((item: any, idx: number) => {
              const percentage = (item.value / total) * 100
              const angle = (percentage * 360) / 100
              const startAngle = cumulativeAngle
              const endAngle = cumulativeAngle + angle
              cumulativeAngle = endAngle
              const startRad = (startAngle - 90) * Math.PI / 180
              const endRad = (endAngle - 90) * Math.PI / 180
              const x1 = 50 + 45 * Math.cos(startRad)
              const y1 = 50 + 45 * Math.sin(startRad)
              const x2 = 50 + 45 * Math.cos(endRad)
              const y2 = 50 + 45 * Math.sin(endRad)
              const largeArcFlag = angle > 180 ? 1 : 0
              const pathData = `M 50 50 L ${x1} ${y1} A 45 45 0 ${largeArcFlag} 1 ${x2} ${y2} Z`
              return <path key={idx} d={pathData} fill={item.color || DEFAULT_COLORS[idx % DEFAULT_COLORS.length]} stroke="white" strokeWidth="1.5" />
            })}
          </svg>
        </div>
        <div className="flex flex-wrap justify-center gap-2 mt-3">
          {data.map((item: any, idx: number) => (
            <div key={idx} className="flex items-center gap-1">
              <div className="w-2.5 h-2.5 rounded-full" style={{ backgroundColor: item.color || DEFAULT_COLORS[idx % DEFAULT_COLORS.length] }} />
              <span className="text-xs text-gray-600">{item.label} ({((item.value / total) * 100).toFixed(0)}%)</span>
            </div>
          ))}
        </div>
      </div>
    </div>
  )
}

// Componente de Gráfico de Linha Moderno com tooltip e linha suave
const GraficoLinhaModerno = ({ periodo, setPeriodo, indicePeriodo, setIndicePeriodo, periodos, labels, series }: any) => {
  const [tooltip, setTooltip] = useState<{ x: number; y: number; label: string; values: any; index: number } | null>(null)
  const svgRef = useRef<SVGSVGElement>(null)
  
  const maxValue = Math.max(...series.flatMap((s: any) => s.data))
  const height = 320
  const width = 900
  const padding = { top: 25, right: 35, bottom: 35, left: 45 }
  const chartHeight = height - padding.top - padding.bottom
  const chartWidth = width - padding.left - padding.right
  
  const getY = (value: number) => padding.top + chartHeight - (value / maxValue) * chartHeight
  const getXPos = (index: number) => padding.left + (index / (labels.length - 1)) * chartWidth
  
  const getSmoothPath = (dataPoints: number[]) => {
    if (dataPoints.length < 2) return ''
    let path = `M ${getXPos(0)} ${getY(dataPoints[0])}`
    for (let i = 1; i < dataPoints.length; i++) {
      const prevX = getXPos(i - 1)
      const prevY = getY(dataPoints[i - 1])
      const currX = getXPos(i)
      const currY = getY(dataPoints[i])
      const cp1x = prevX + (currX - prevX) * 0.3
      const cp1y = prevY
      const cp2x = currX - (currX - prevX) * 0.3
      const cp2y = currY
      path += ` C ${cp1x} ${cp1y}, ${cp2x} ${cp2y}, ${currX} ${currY}`
    }
    return path
  }
  
  const handleMouseMove = (e: React.MouseEvent<SVGSVGElement>, idx: number, values: any) => {
    const rect = e.currentTarget.getBoundingClientRect()
    const mouseX = e.clientX - rect.left
    const mouseY = e.clientY - rect.top
    setTooltip({
      x: mouseX,
      y: mouseY - 40,
      label: labels[idx],
      values: values,
      index: idx
    })
  }
  
  const handleMouseLeave = () => {
    setTooltip(null)
  }
  
  const handlePeriodoAnterior = () => {
    if (indicePeriodo < periodos.length - 1) setIndicePeriodo(indicePeriodo + 1)
  }
  
  const handlePeriodoProximo = () => {
    if (indicePeriodo > 0) setIndicePeriodo(indicePeriodo - 1)
  }

  const getPeriodoNome = () => {
    if (periodo === 'semana') return 'Semanal (dados diários)'
    if (periodo === 'mes') return 'Mensal (dados semanais)'
    return 'Anual (dados mensais)'
  }

  return (
    <div className="bg-white rounded-xl shadow-md p-5">
      <div className="flex flex-wrap justify-between items-center gap-3 mb-4">
        <h3 className="font-semibold text-gray-800 text-base">Crescimento de Usuários</h3>
        <div className="flex items-center gap-2">
          <div className="flex items-center gap-1 bg-gray-100 rounded-lg p-0.5">
            <button onClick={handlePeriodoAnterior} className="p-1.5 hover:bg-white rounded-md transition text-gray-600"><ChevronLeft size={16} /></button>
            <div className="relative">
              <select value={periodo} onChange={(e) => { setPeriodo(e.target.value); setIndicePeriodo(0) }} className="px-3 py-1.5 border-0 bg-transparent text-sm text-gray-700 font-medium appearance-none cursor-pointer pr-6 focus:outline-none">
                <option value="semana">📊 Semanal</option>
                <option value="mes">📈 Mensal</option>
                <option value="ano">📉 Anual</option>
              </select>
              <ChevronDown size={14} className="absolute right-1 top-1/2 transform -translate-y-1/2 text-gray-400 pointer-events-none" />
            </div>
            <button onClick={handlePeriodoProximo} disabled={indicePeriodo === 0} className={`p-1.5 rounded-md transition ${indicePeriodo === 0 ? 'text-gray-300 cursor-not-allowed' : 'hover:bg-white text-gray-600'}`}><ChevronRight size={16} /></button>
          </div>
        </div>
      </div>
      <p className="text-xs text-gray-500 mb-4 text-center">{periodos[indicePeriodo]?.nome || 'Período atual'} • {getPeriodoNome()}</p>
      <div className="relative overflow-x-auto">
        <svg ref={svgRef} className="w-full" viewBox={`0 0 ${width} ${height}`} preserveAspectRatio="xMidYMid meet" style={{ cursor: 'crosshair' }}>
          {[0, 20, 40, 60, 80, 100].map((p, i) => {
            const y = getY(maxValue * p / 100)
            return (
              <g key={i} className="opacity-30">
                <line x1={padding.left} y1={y} x2={width - padding.right} y2={y} stroke="#D1D5DB" strokeWidth="0.8" strokeDasharray="3 3" />
                <text x={padding.left - 8} y={y + 3} fontSize="10" fill="#9CA3AF" textAnchor="end">{Math.round(maxValue * p / 100)}</text>
              </g>
            )
          })}
          {series.map((serie: any, idx: number) => (
            <path key={`area-${idx}`} d={`${getSmoothPath(serie.data)} L ${getXPos(labels.length - 1)} ${getY(0)} L ${getXPos(0)} ${getY(0)} Z`} fill={serie.color} fillOpacity="0.06" className="transition-all duration-500" />
          ))}
          {series.map((serie: any, idx: number) => (
            <path key={`line-${idx}`} d={getSmoothPath(serie.data)} fill="none" stroke={serie.color} strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" className="transition-all duration-500" />
          ))}
          {labels.map((label: string, i: number) => {
            const valuesAtPoint: any = {}
            series.forEach((serie: any) => { valuesAtPoint[serie.name] = serie.data[i] })
            const allPoints = series.flatMap(s => s.data[i])
            const maxPoint = Math.max(...allPoints)
            return (
              <g key={`point-group-${i}`} onMouseMove={(e) => handleMouseMove(e, i, valuesAtPoint)} onMouseLeave={handleMouseLeave} className="cursor-pointer">
                {series.map((serie: any, sIdx: number) => {
                  const isHighest = serie.data[i] === maxPoint
                  return <circle key={`point-${sIdx}-${i}`} cx={getXPos(i)} cy={getY(serie.data[i])} r={isHighest ? 5 : 3.5} fill={serie.color} stroke="white" strokeWidth="2" className="transition-all duration-300 hover:r-6" />
                })}
              </g>
            )
          })}
          {labels.map((label: string, i: number) => (
            <text key={`label-${i}`} x={getXPos(i)} y={height - padding.bottom + 15} fontSize="10" fill="#6B7280" textAnchor="middle" className="font-medium">{label}</text>
          ))}
        </svg>
        {tooltip && (
          <div className="fixed bg-gray-900 text-white rounded-lg shadow-xl p-3 z-50 pointer-events-none" style={{ left: tooltip.x + 15, top: tooltip.y - 10, transform: 'translate(-50%, -100%)', minWidth: '200px' }}>
            <div className="text-xs font-semibold text-gray-300 border-b border-gray-700 pb-1.5 mb-1.5">{tooltip.label}</div>
            {Object.entries(tooltip.values).map(([name, value]: [string, any]) => {
              let color = '#3B82F6'
              if (name === 'Empresas') color = '#10B981'
              else if (name === 'Ativos') color = '#F59E0B'
              else if (name === 'Academia') color = '#8B5CF6'
              return (
                <div key={name} className="flex items-center justify-between gap-6 text-xs py-1">
                  <div className="flex items-center gap-2"><div className="w-2.5 h-2.5 rounded-full" style={{ backgroundColor: color }} /><span>{name}</span></div>
                  <span className="font-mono font-semibold">{value.toLocaleString()}</span>
                </div>
              )
            })}
          </div>
        )}
      </div>
      <div className="flex flex-wrap justify-center gap-6 mt-6 pt-4 border-t border-gray-100">
        {series.map((serie: any, idx: number) => {
          const ultimoValor = serie.data[serie.data.length - 1]
          const primeiroValor = serie.data[0]
          const variacao = ultimoValor - primeiroValor
          return (
            <div key={idx} className="flex items-center gap-2 group cursor-pointer hover:scale-105 transition-transform">
              <div className="w-3 h-3 rounded-full" style={{ backgroundColor: serie.color }} />
              <span className="text-sm text-gray-600 group-hover:text-gray-800">{serie.name}</span>
              <span className={`text-xs font-semibold ${variacao >= 0 ? 'text-green-600' : 'text-red-600'}`}>{variacao >= 0 ? '+' : ''}{variacao}</span>
            </div>
          )
        })}
      </div>
    </div>
  )
}

// Componente de Gráfico de Pizza
const GraficoPizza = ({ data, title }: any) => {
  const total = data.reduce((sum: number, item: any) => sum + item.value, 0)
  const DEFAULT_COLORS = ['#3B82F6', '#10B981', '#F59E0B', '#EF4444', '#8B5CF6', '#EC4899']
  let cumulativeAngle = 0
  return (
    <div className="bg-white rounded-xl shadow-md p-5">
      <h3 className="font-semibold text-gray-700 mb-4">{title}</h3>
      <div className="flex flex-col items-center">
        <div className="relative w-40 h-40">
          <svg viewBox="0 0 100 100" className="w-full h-full">
            {data.map((item: any, idx: number) => {
              const percentage = (item.value / total) * 100
              const angle = (percentage * 360) / 100
              const startAngle = cumulativeAngle
              const endAngle = cumulativeAngle + angle
              cumulativeAngle = endAngle
              const startRad = (startAngle - 90) * Math.PI / 180
              const endRad = (endAngle - 90) * Math.PI / 180
              const x1 = 50 + 45 * Math.cos(startRad)
              const y1 = 50 + 45 * Math.sin(startRad)
              const x2 = 50 + 45 * Math.cos(endRad)
              const y2 = 50 + 45 * Math.sin(endRad)
              const largeArcFlag = angle > 180 ? 1 : 0
              const pathData = `M 50 50 L ${x1} ${y1} A 45 45 0 ${largeArcFlag} 1 ${x2} ${y2} Z`
              return <path key={idx} d={pathData} fill={item.color || DEFAULT_COLORS[idx % DEFAULT_COLORS.length]} stroke="white" strokeWidth="1.5" />
            })}
          </svg>
        </div>
        <div className="flex flex-wrap justify-center gap-2 mt-3">
          {data.map((item: any, idx: number) => (
            <div key={idx} className="flex items-center gap-1">
              <div className="w-2.5 h-2.5 rounded-full" style={{ backgroundColor: item.color || DEFAULT_COLORS[idx % DEFAULT_COLORS.length] }} />
              <span className="text-xs text-gray-600">{item.label} ({((item.value / total) * 100).toFixed(0)}%)</span>
            </div>
          ))}
        </div>
      </div>
    </div>
  )
}

'use client'

import { useState, useEffect } from 'react'
import { supabase } from '@/lib/supabaseClient'

export default function FinanceiroPessoalScreen() {
  const [lancamentos, setLancamentos] = useState<any[]>([])
  const [saldo, setSaldo] = useState(0)
  const [loading, setLoading] = useState(true)
  const [showModal, setShowModal] = useState(false)
  const [editando, setEditando] = useState<any>(null)
  const [form, setForm] = useState({
    descricao: '',
    valor: '',
    tipo: 'despesa',
    categoria: '',
    data: new Date().toISOString().split('T')[0]
  })

  useEffect(() => {
    carregarDados()
  }, [])

  async function carregarDados() {
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) {
      setLoading(false)
      return
    }

    const { data: lancamentosData } = await supabase
      .from('financas_pessoais')
      .select('*')
      .eq('usuario_id', user.id)
      .order('data', { ascending: false })

    setLancamentos(lancamentosData || [])
    
    const totalReceitas = lancamentosData?.filter(l => l.tipo === 'receita').reduce((acc, l) => acc + l.valor, 0) || 0
    const totalDespesas = lancamentosData?.filter(l => l.tipo === 'despesa').reduce((acc, l) => acc + l.valor, 0) || 0
    setSaldo(totalReceitas - totalDespesas)
    
    setLoading(false)
  }

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault()
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) return

    if (editando) {
      const { error } = await supabase
        .from('financas_pessoais')
        .update({
          descricao: form.descricao,
          valor: parseFloat(form.valor),
          tipo: form.tipo,
          categoria: form.categoria,
          data: form.data
        })
        .eq('id', editando.id)
      
      if (!error) {
        setShowModal(false)
        setEditando(null)
        setForm({ descricao: '', valor: '', tipo: 'despesa', categoria: '', data: new Date().toISOString().split('T')[0] })
        carregarDados()
      }
    } else {
      const { error } = await supabase.from('financas_pessoais').insert({
        usuario_id: user.id,
        descricao: form.descricao,
        valor: parseFloat(form.valor),
        tipo: form.tipo,
        categoria: form.categoria,
        data: form.data
      })
      
      if (!error) {
        setShowModal(false)
        setForm({ descricao: '', valor: '', tipo: 'despesa', categoria: '', data: new Date().toISOString().split('T')[0] })
        carregarDados()
      }
    }
  }

  function handleEditar(lancamento: any) {
    setEditando(lancamento)
    setForm({
      descricao: lancamento.descricao,
      valor: lancamento.valor.toString(),
      tipo: lancamento.tipo,
      categoria: lancamento.categoria || '',
      data: lancamento.data
    })
    setShowModal(true)
  }

  async function handleExcluir(id: string) {
    if (confirm('Tem certeza que deseja excluir este lançamento?')) {
      await supabase.from('financas_pessoais').delete().eq('id', id)
      carregarDados()
    }
  }

  if (loading) {
    return <div className="p-8 text-center">Carregando...</div>
  }

  return (
    <div className="p-6 max-w-6xl mx-auto">
      <div className="flex justify-between items-center mb-6">
        <h1 className="text-2xl font-bold">💰 Finanças Pessoais</h1>
        <button onClick={() => { setEditando(null); setForm({ descricao: '', valor: '', tipo: 'despesa', categoria: '', data: new Date().toISOString().split('T')[0] }); setShowModal(true) }} className="bg-blue-600 text-white px-4 py-2 rounded-lg hover:bg-blue-700">
          + Novo Lançamento
        </button>
      </div>
      
      <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-8">
        <div className="bg-white rounded-lg shadow p-4">
          <div className="text-gray-500 text-sm">Saldo Total</div>
          <div className={`text-2xl font-bold ${saldo >= 0 ? 'text-green-600' : 'text-red-600'}`}>
            R$ {saldo.toFixed(2)}
          </div>
        </div>
        <div className="bg-white rounded-lg shadow p-4">
          <div className="text-gray-500 text-sm">Receitas</div>
          <div className="text-2xl font-bold text-green-600">
            R$ {lancamentos.filter(l => l.tipo === 'receita').reduce((acc, l) => acc + l.valor, 0).toFixed(2)}
          </div>
        </div>
        <div className="bg-white rounded-lg shadow p-4">
          <div className="text-gray-500 text-sm">Despesas</div>
          <div className="text-2xl font-bold text-red-600">
            R$ {lancamentos.filter(l => l.tipo === 'despesa').reduce((acc, l) => acc + l.valor, 0).toFixed(2)}
          </div>
        </div>
      </div>

      <div className="bg-white rounded-lg shadow overflow-hidden">
        <div className="overflow-x-auto">
          <table className="w-full">
            <thead className="bg-gray-50">
              <tr>
                <th className="px-4 py-3 text-left">Data</th>
                <th className="px-4 py-3 text-left">Descrição</th>
                <th className="px-4 py-3 text-left">Categoria</th>
                <th className="px-4 py-3 text-right">Valor</th>
                <th className="px-4 py-3 text-center">Tipo</th>
                <th className="px-4 py-3 text-center">Ações</th>
              </tr>
            </thead>
            <tbody className="divide-y">
              {lancamentos.map((l) => (
                <tr key={l.id}>
                  <td className="px-4 py-3">{new Date(l.data).toLocaleDateString()}</td>
                  <td className="px-4 py-3">{l.descricao}</td>
                  <td className="px-4 py-3">{l.categoria || '-'}</td>
                  <td className={`px-4 py-3 text-right font-medium ${l.tipo === 'receita' ? 'text-green-600' : 'text-red-600'}`}>
                    R$ {l.valor.toFixed(2)}
                  </td>
                  <td className="px-4 py-3 text-center">
                    <span className={`px-2 py-1 rounded-full text-xs ${l.tipo === 'receita' ? 'bg-green-100 text-green-800' : 'bg-red-100 text-red-800'}`}>
                      {l.tipo === 'receita' ? 'Receita' : 'Despesa'}
                    </span>
                  </td>
                  <td className="px-4 py-3 text-center">
                    <button onClick={() => handleEditar(l)} className="text-blue-600 hover:text-blue-800 mr-2">Editar</button>
                    <button onClick={() => handleExcluir(l.id)} className="text-red-600 hover:text-red-800">Excluir</button>
                  </td>
                </tr>
              ))}
              {lancamentos.length === 0 && (
                <tr>
                  <td colSpan={6} className="px-4 py-8 text-center text-gray-500">
                    Nenhum lançamento encontrado.
                  </td>
                </tr>
              )}
            </tbody>
          </table>
        </div>
      </div>

      {/* Modal */}
      {showModal && (
        <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50 p-4">
          <div className="bg-white rounded-xl max-w-md w-full p-6">
            <h2 className="text-xl font-bold mb-4">{editando ? 'Editar' : 'Novo'} Lançamento</h2>
            <form onSubmit={handleSubmit} className="space-y-4">
              <input type="text" placeholder="Descrição" value={form.descricao} onChange={(e) => setForm({ ...form, descricao: e.target.value })} className="w-full border rounded-lg px-3 py-2" required />
              <input type="number" step="0.01" placeholder="Valor" value={form.valor} onChange={(e) => setForm({ ...form, valor: e.target.value })} className="w-full border rounded-lg px-3 py-2" required />
              <select value={form.tipo} onChange={(e) => setForm({ ...form, tipo: e.target.value })} className="w-full border rounded-lg px-3 py-2">
                <option value="despesa">Despesa</option>
                <option value="receita">Receita</option>
              </select>
              <input type="text" placeholder="Categoria" value={form.categoria} onChange={(e) => setForm({ ...form, categoria: e.target.value })} className="w-full border rounded-lg px-3 py-2" />
              <input type="date" value={form.data} onChange={(e) => setForm({ ...form, data: e.target.value })} className="w-full border rounded-lg px-3 py-2" />
              <div className="flex gap-3">
                <button type="submit" className="flex-1 bg-blue-600 text-white py-2 rounded-lg hover:bg-blue-700">Salvar</button>
                <button type="button" onClick={() => { setShowModal(false); setEditando(null) }} className="flex-1 border border-gray-300 py-2 rounded-lg hover:bg-gray-50">Cancelar</button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  )
}
3. Criar a tabela no Supabase

sql
CREATE TABLE IF NOT EXISTS financas_pessoais (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  usuario_id UUID NOT NULL,
  descricao TEXT NOT NULL,
  valor DECIMAL(10,2) NOT NULL,
  tipo VARCHAR(20) NOT NULL CHECK (tipo IN ('receita', 'despesa')),
  categoria VARCHAR(100),
  data DATE NOT NULL,
  created_at TIMESTAMP DEFAULT NOW()
);
