import { useState, useEffect, useCallback } from 'react';
import { supabase } from '@/lib/supabase';

export interface Producao {
  id: string;
  prato_id: string;
  data_producao: string;
  quantidade_prevista: number;
  quantidade_produzida: number;
  status: 'planejado' | 'em_producao' | 'concluido' | 'cancelado';
  observacoes: string;
  created_at: string;
  updated_at: string;
}

export function useProducao() {
  const [producoes, setProducoes] = useState<Producao[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const fetchProducoes = useCallback(async () => {
    try {
      setLoading(true);
      const { data, error } = await supabase
        .from('producao')
        .select('*')
        .order('data_producao', { ascending: false });

      if (error) throw error;
      setProducoes(data || []);
      setError(null);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Erro ao carregar produÃ§Ãµes');
    } finally {
      setLoading(false);
    }
  }, []);

  const createProducao = useCallback(async (producao: Omit<Producao, 'id' | 'created_at' | 'updated_at'>) => {
    try {
      setLoading(true);
      const { data, error } = await supabase
        .from('producao')
        .insert([producao])
        .select()
        .single();

      if (error) throw error;
      await fetchProducoes();
      return data;
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Erro ao criar produÃ§Ã£o');
      return null;
    } finally {
      setLoading(false);
    }
  }, [fetchProducoes]);

  const updateProducao = useCallback(async (id: string, updates: Partial<Producao>) => {
    try {
      setLoading(true);
      const { data, error } = await supabase
        .from('producao')
        .update(updates)
        .eq('id', id)
        .select()
        .single();

      if (error) throw error;
      await fetchProducoes();
      return data;
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Erro ao atualizar produÃ§Ã£o');
      return null;
    } finally {
      setLoading(false);
    }
  }, [fetchProducoes]);

  const deleteProducao = useCallback(async (id: string) => {
    try {
      setLoading(true);
      const { error } = await supabase
        .from('producao')
        .delete()
        .eq('id', id);

      if (error) throw error;
      await fetchProducoes();
      return true;
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Erro ao deletar produÃ§Ã£o');
      return false;
    } finally {
      setLoading(false);
    }
  }, [fetchProducoes]);

  const getProducaoPorData = useCallback((data: string) => {
    return producoes.filter(prod => prod.data_producao === data);
  }, [producoes]);

  useEffect(() => {
    fetchProducoes();
  }, [fetchProducoes]);

  return {
    producoes,
    loading,
    error,
    fetchProducoes,
    createProducao,
    updateProducao,
    deleteProducao,
    getProducaoPorData,
  };
}
