-- ============================================================================
-- TABELA: configuracoes_cambio (Controle de câmbio por cidade)
-- ============================================================================
CREATE TABLE IF NOT EXISTS configuracoes_cambio (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  cidade TEXT NOT NULL UNIQUE,
  estado TEXT NOT NULL,
  taxa_cambio DECIMAL(10,4) DEFAULT 1.0,
  taxa_compra DECIMAL(10,4) DEFAULT 1.0,
  taxa_venda DECIMAL(10,4) DEFAULT 1.0,
  atualizado_em TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  atualizado_por TEXT,
  ativo BOOLEAN DEFAULT true
);
