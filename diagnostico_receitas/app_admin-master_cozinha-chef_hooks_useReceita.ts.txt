// hooks/cozinha/useReceita.ts

import { useState, useEffect, useCallback } from 'react';
import { Receita, IngredienteReceita, TotaisReceita } from '@/types/cozinha';
import { ReceitaCanonicaCompat } from '@/types/receita-canonica';
import { calcularTotaisReceita, calcularPorcoes } from '@/utils/cozinhaUtils';
import {
  buildReceitaPayload,
  calcularFinanceiroReceita,
  normalizarIngredienteCanonico,
  normalizarReceitaCanonica as normalizarReceitaCanonicaOficial,
} from '../services/custoService';

type ApiResult<T> = {
  success?: boolean;
  data?: T;
  error?: string;
};

const DEFAULT_INTEGRACOES = {
  catalogo: false,
  cardapio: false,
  producao: false,
  estoque: false,
  compras: false,
};

function asNumber(value: unknown, fallback = 0): number {
  const n = Number(value);
  return Number.isFinite(n) ? n : fallback;
}

function extractData<T>(payload: any): T {
  if (payload && typeof payload === 'object' && 'success' in payload) {
    return ((payload as ApiResult<T>).data as T) ?? ([] as unknown as T);
  }
  return (payload ?? ([] as unknown as T)) as T;
}

function normalizeReceitaCanonica(item: any): ReceitaCanonicaCompat {
  return normalizarReceitaCanonicaOficial(item);
}

function toLegacyReceita(canonica: ReceitaCanonicaCompat): Receita {
  return {
    id: canonica.id,
    nome: canonica.nome,
    descricao: canonica.descricao,
    categoria: canonica.categoria,
    porcoes: asNumber(canonica.porcoes, 0),
    preco_sugerido: asNumber(canonica.preco_sugerido, 0),
    custo_total: asNumber(canonica.custo_receita ?? canonica.custo_total, 0),
    ingredientes: (canonica.ingredientes || []).map((ing) => ({
      ingrediente_nome: ing.ingrediente_nome,
      ingrediente_id: ing.ingrediente_id,
      quantidade: asNumber(ing.quantidade, 0),
      unidade: ing.unidade || 'un',
      custo_total: asNumber(ing.custo_total, 0),
    })),
    ativo: canonica.status === 'ativo',
    created_at: new Date(canonica.created_at),
    updated_at: new Date(canonica.updated_at),
  };
}

function toCanonicalFromLegacy(legacy: Receita): ReceitaCanonicaCompat {
  const ingredientes = Array.isArray(legacy.ingredientes)
    ? legacy.ingredientes.map((ing) =>
      normalizarIngredienteCanonico({
        ingrediente_id: ing.ingrediente_id,
        ingrediente_nome: ing.ingrediente_nome,
        quantidade: ing.quantidade,
        unidade: ing.unidade,
        custo_total: ing.custo_total,
        custo_unitario: asNumber(ing.custo_total, 0) / Math.max(asNumber(ing.quantidade, 0), 1),
      })
    )
    : [];

  const financeiro = calcularFinanceiroReceita({
    ingredientes,
    porcoes: legacy.porcoes,
    preco_venda: legacy.preco_sugerido,
    preco_sugerido: legacy.preco_sugerido,
    custo_receita: legacy.custo_total,
  });

  return {
    id: String(legacy.id || ''),
    nome: String(legacy.nome || ''),
    descricao: String(legacy.descricao || ''),
    categoria: String(legacy.categoria || 'Geral'),
    imagem: null,
    status: legacy.ativo ? 'ativo' : 'inativo',
    ingredientes,
    rendimento: asNumber(legacy.porcoes, 0),
    peso_final: null,
    porcoes: financeiro.porcoes,
    custo_receita: financeiro.custo_receita,
    custo_por_unidade: financeiro.custo_por_unidade,
    margem_percentual: financeiro.margem_percentual,
    lucro: financeiro.lucro,
    preco_sugerido: financeiro.preco_sugerido,
    preco_venda: financeiro.preco_venda,
    integracoes: { ...DEFAULT_INTEGRACOES },
    created_at: legacy.created_at?.toISOString?.() ?? new Date().toISOString(),
    updated_at: legacy.updated_at?.toISOString?.() ?? new Date().toISOString(),
    preco: financeiro.preco_venda,
    custo_total: financeiro.custo_receita,
    margem: financeiro.margem_percentual,
    ativo: legacy.ativo,
    servings: financeiro.porcoes,
    isAvailable: legacy.ativo,
  };
}

function toApiPayload(canonica: ReceitaCanonicaCompat) {
  return buildReceitaPayload(canonica);
}

export function useReceita(receitaId?: string) {
  // ============================================================
  // ESTADOS
  // ============================================================
  const [receitaCanonica, setReceitaCanonica] = useState<ReceitaCanonicaCompat | null>(null);
  const [receitasCanonicas, setReceitasCanonicas] = useState<ReceitaCanonicaCompat[]>([]);
  const [totais, setTotais] = useState<TotaisReceita | null>(null);
  const [loading, setLoading] = useState(true);
  const [salvando, setSalvando] = useState(false);
  const [quantidadePorcoes, setQuantidadePorcoes] = useState(1);
  const [ingredientesDisponiveis, setIngredientesDisponiveis] = useState<any[]>([]);
  const [pesoMeta, setPesoMeta] = useState<Record<string, number>>({});
  const [error, setError] = useState<string | null>(null);

  // ============================================================
  // EFECTS
  // ============================================================
  
  // Calcular totais quando receita mudar
  useEffect(() => {
    if (receitaCanonica) {
      const totaisCalculados = calcularTotaisReceita(toLegacyReceita(receitaCanonica));
      setTotais(totaisCalculados);
      
      // Calcular metas de peso (10% a mais para margem de seguranÃ§a)
      const metas: Record<string, number> = {};
      (receitaCanonica.ingredientes || []).forEach(ing => {
        metas[ing.ingrediente_nome] = ing.quantidade * 1.1;
      });
      setPesoMeta(metas);
    }
  }, [receitaCanonica]);

  // Carregar na montagem
  useEffect(() => {
    if (receitaId) {
      carregarReceita();
    }
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [receitaId]);

  // ============================================================
  // FUNCOES DE CARREGAMENTO
  // ============================================================

  // Carregar ingredientes disponiveis
  const carregarIngredientes = useCallback(async () => {
    try {
      const response = await fetch('/api/cozinha/estoque');
      const payload = await response.json();
      const data = extractData<any[]>(payload);

      const ingredientes = (Array.isArray(data) ? data : []).map((item) => ({
        id: String(item?.id ?? ''),
        nome: String(item?.nome ?? item?.produto ?? ''),
        preco_unitario: asNumber(item?.preco_unitario, 0),
        unidade: String(item?.unidade ?? 'un'),
      }));

      setIngredientesDisponiveis(ingredientes);
    } catch (error) {
      console.error('Erro ao carregar ingredientes:', error);
      setIngredientesDisponiveis([]);
    }
  }, []);

  const listarReceitas = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      let response = await fetch('/api/cozinha/receitas');
      let payload = await response.json();
      let data = extractData<any[]>(payload);

      if (!Array.isArray(data) || data.length === 0) {
        response = await fetch('/api/cozinha/recipes');
        payload = await response.json();
        data = extractData<any[]>(payload);
      }

      const normalized = (Array.isArray(data) ? data : []).map(normalizeReceitaCanonica);
      setReceitasCanonicas(normalized);
      return normalized;
    } catch (err) {
      const message = err instanceof Error ? err.message : 'Erro ao listar receitas';
      setError(message);
      setReceitasCanonicas([]);
      return [] as ReceitaCanonicaCompat[];
    } finally {
      setLoading(false);
    }
  }, []);

  // Carregar receita
  const carregarReceita = useCallback(async () => {
    if (!receitaId) return;
    
    setLoading(true);
    setError(null);
    
    try {
      let response = await fetch(`/api/cozinha/receitas/${encodeURIComponent(receitaId)}`);
      let payload = await response.json();
      let receitaData = extractData<any>(payload);

      if (!receitaData || !receitaData.id) {
        response = await fetch(`/api/cozinha/receitas?id=${encodeURIComponent(receitaId)}`);
        payload = await response.json();
        receitaData = extractData<any>(payload);
      }

      if (!receitaData || !receitaData.id) {
        response = await fetch('/api/cozinha/recipes');
        payload = await response.json();
        const recipes = extractData<any[]>(payload);
        receitaData = (Array.isArray(recipes) ? recipes : []).find((r: any) => String(r?.id) === String(receitaId));
      }

      if (!receitaData) {
        setError('Receita nao encontrada');
        setReceitaCanonica(null);
        return;
      }

      setReceitaCanonica(normalizeReceitaCanonica(receitaData));
      await carregarIngredientes();
    } catch (error) {
      console.error('Erro ao carregar receita:', error);
      setError('Erro ao carregar receita');
      setReceitaCanonica(null);
    } finally {
      setLoading(false);
    }
  }, [receitaId, carregarIngredientes]);

  // ============================================================
  // FUNCOES DE SALVAMENTO
  // ============================================================

  // Salvar receita (retorna resultado)
  const salvarReceita = useCallback(async (): Promise<{ success: boolean; data?: Receita; error?: string }> => {
    if (!receitaCanonica) {
      return { success: false, error: 'Nenhuma receita para salvar' };
    }
    
    setSalvando(true);
    setError(null);
    
    try {
      const payload = toApiPayload({
        ...receitaCanonica,
        updated_at: new Date().toISOString(),
      });

      const response = receitaCanonica.id
        ? await fetch(`/api/cozinha/receitas/${encodeURIComponent(receitaCanonica.id)}`, {
          method: 'PUT',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload),
        })
        : await fetch('/api/cozinha/receitas', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload),
        });

      let result = await response.json();

      if (!response.ok || result?.success === false) {
        // fallback para endpoint alternativo com query param
        if (receitaCanonica.id) {
          const fallbackResponse = await fetch(`/api/cozinha/receitas?id=${encodeURIComponent(receitaCanonica.id)}`, {
            method: 'PUT',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload),
          });
          result = await fallbackResponse.json();
          if (!fallbackResponse.ok || result?.success === false) {
            throw new Error(result?.error || 'Erro ao salvar receita');
          }
        } else {
          throw new Error(result?.error || 'Erro ao criar receita');
        }
      }

      const saved = normalizeReceitaCanonica(extractData<any>(result));
      setReceitaCanonica(saved);

      return { success: true, data: toLegacyReceita(saved) };
    } catch (error: any) {
      console.error('Erro ao salvar receita:', error);
      setError(error.message || 'Erro ao salvar receita');
      return { success: false, error: error.message || 'Erro ao salvar receita' };
    } finally {
      setSalvando(false);
    }
  }, [receitaCanonica]);

  const criarReceita = useCallback(async (dados: Partial<ReceitaCanonicaCompat>) => {
    setSalvando(true);
    setError(null);
    try {
      const base: ReceitaCanonicaCompat = normalizeReceitaCanonica({
        ...dados,
        id: '',
        status: dados.status || 'inativo',
      });

      const response = await fetch('/api/cozinha/receitas', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(toApiPayload(base)),
      });

      const result = await response.json();
      if (!response.ok || result?.success === false) {
        throw new Error(result?.error || 'Erro ao criar receita');
      }

      const created = normalizeReceitaCanonica(extractData<any>(result));
      setReceitaCanonica(created);
      return { success: true, data: created };
    } catch (err: any) {
      setError(err?.message || 'Erro ao criar receita');
      return { success: false, error: err?.message || 'Erro ao criar receita' };
    } finally {
      setSalvando(false);
    }
  }, []);

  const excluirReceita = useCallback(async (id: string) => {
    if (!id) return { success: false, error: 'ID invalido' };

    setSalvando(true);
    setError(null);
    try {
      let response = await fetch(`/api/cozinha/receitas/${encodeURIComponent(id)}`, { method: 'DELETE' });
      let result = await response.json();

      if (!response.ok || result?.success === false) {
        response = await fetch(`/api/cozinha/receitas?id=${encodeURIComponent(id)}`, { method: 'DELETE' });
        result = await response.json();
        if (!response.ok || result?.success === false) {
          throw new Error(result?.error || 'Erro ao excluir receita');
        }
      }

      if (receitaCanonica?.id === id) {
        setReceitaCanonica(null);
      }
      setReceitasCanonicas((prev) => prev.filter((r) => r.id !== id));
      return { success: true };
    } catch (err: any) {
      setError(err?.message || 'Erro ao excluir receita');
      return { success: false, error: err?.message || 'Erro ao excluir receita' };
    } finally {
      setSalvando(false);
    }
  }, [receitaCanonica?.id]);

  // ============================================================
  // FUNCOES DE MANIPULACAO DE INGREDIENTES
  // ============================================================

  // Adicionar ingrediente
  const adicionarIngrediente = useCallback((ingrediente: IngredienteReceita) => {
    if (!receitaCanonica) return;

    const quantidade = asNumber(ingrediente.quantidade, 0);
    const custoTotal = asNumber(ingrediente.custo_total, 0);
    const custoUnitario = quantidade > 0 ? custoTotal / quantidade : 0;

    const ingredienteCanonico = normalizarIngredienteCanonico({
      ingrediente_id: ingrediente.ingrediente_id,
      ingrediente_nome: ingrediente.ingrediente_nome,
      quantidade,
      unidade: ingrediente.unidade,
      custo_unitario: custoUnitario,
      custo_total: custoTotal,
    });

    setReceitaCanonica({
      ...receitaCanonica,
      ingredientes: [
        ...(receitaCanonica.ingredientes || []),
        ingredienteCanonico,
      ],
    });
  }, [receitaCanonica]);

  // Remover ingrediente
  const removerIngrediente = useCallback((index: number) => {
    if (!receitaCanonica) return;

    const novosIngredientes = (receitaCanonica.ingredientes || []).filter((_, i) => i !== index);
    setReceitaCanonica({
      ...receitaCanonica,
      ingredientes: novosIngredientes
    });
  }, [receitaCanonica]);

  // Atualizar quantidade de ingrediente
  const atualizarQuantidade = useCallback((index: number, quantidade: number) => {
    if (!receitaCanonica) return;

    const novosIngredientes = (receitaCanonica.ingredientes || []).map((ing, i) => {
      if (i === index) {
        const baseQuantidade = Math.max(asNumber(ing.quantidade, 0), 1);
        const proporcao = asNumber(quantidade, 0) / baseQuantidade;
        return {
          ...ing,
          quantidade: asNumber(quantidade, 0),
          custo_total: asNumber(ing.custo_total, 0) * proporcao,
        };
      }
      return ing;
    });

    setReceitaCanonica({
      ...receitaCanonica,
      ingredientes: novosIngredientes
    });
  }, [receitaCanonica]);

  // ============================================================
  // FUNCOES DE CALCULO
  // ============================================================

  // Calcular porcoes
  const calcularPorcoesReceita = useCallback(() => {
    if (!receitaCanonica) return null;
    return calcularPorcoes(toLegacyReceita(receitaCanonica), quantidadePorcoes);
  }, [receitaCanonica, quantidadePorcoes]);

  // Calcular margem
  const calcularMargem = useCallback(() => {
    if (!receitaCanonica) return 0;
    const financeiro = calcularFinanceiroReceita({
      ingredientes: receitaCanonica.ingredientes,
      porcoes: receitaCanonica.porcoes,
      preco_venda: receitaCanonica.preco_venda,
      preco_sugerido: receitaCanonica.preco_sugerido,
      custo_receita: receitaCanonica.custo_receita,
    });
    return financeiro.margem_percentual;
  }, [receitaCanonica]);

  // ============================================================
  // FUNCOES DE RESET
  // ============================================================

  // Resetar formulario
  const resetForm = useCallback(() => {
    setReceitaCanonica(null);
    setReceitasCanonicas([]);
    setTotais(null);
    setError(null);
    setQuantidadePorcoes(1);
    setPesoMeta({});
    setIngredientesDisponiveis([]);
  }, []);

  const receita = receitaCanonica ? toLegacyReceita(receitaCanonica) : null;
  const receitas = receitasCanonicas.map(toLegacyReceita);

  const setReceita = useCallback((value: Receita | null | ((prev: Receita | null) => Receita | null)) => {
    if (typeof value === 'function') {
      const nextLegacy = value(receita);
      setReceitaCanonica(nextLegacy ? toCanonicalFromLegacy(nextLegacy) : null);
      return;
    }
    setReceitaCanonica(value ? toCanonicalFromLegacy(value) : null);
  }, [receita]);

  // ============================================================
  // RETURN
  // ============================================================

  return {
    // Dados
    receita,
    receitaCanonica,
    receitas,
    receitasCanonicas,
    setReceita,
    setReceitaCanonica,
    totais,
    loading,
    salvando,
    error,
    
    // Porcoes
    quantidadePorcoes,
    setQuantidadePorcoes,
    
    // Ingredientes
    ingredientesDisponiveis,
    pesoMeta,
    
    // Funcoes de calculo
    calcularPorcoes: calcularPorcoesReceita,
    calcularMargem,
    
    // Funcoes CRUD
    salvarReceita,
    criarReceita,
    excluirReceita,
    listarReceitas,
    carregarReceita,
    resetForm,
    
    // Funcoes de manipulacao
    adicionarIngrediente,
    removerIngrediente,
    atualizarQuantidade,
  };
}

export default useReceita;
