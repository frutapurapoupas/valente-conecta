import { supabase } from '@/lib/supabase/client';

export const dashboardService = {
  async getStats() {
    const [estoque, receitas, compras, producao] = await Promise.all([
      supabase.from('estoque').select('*'),
      supabase.from('receitas').select('*'),
      supabase.from('compras').select('*'),
      supabase.from('producao').select('*')
    ]);

    return {
      totalEstoque: estoque.data?.length || 0,
      totalReceitas: receitas.data?.length || 0,
      totalCompras: compras.data?.length || 0,
      totalProducao: producao.data?.length || 0
    };
  }
};
