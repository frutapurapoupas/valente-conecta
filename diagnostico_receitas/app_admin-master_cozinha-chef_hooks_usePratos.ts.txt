import { useState, useEffect, useCallback } from 'react';
import { supabase } from '@/lib/supabase';

export interface Prato {
  id: string;
  nome: string;
  descricao: string;
  categoria: string;
  preco: number;
  tempo_preparo: number;
  ingredientes: string[];
  status: 'ativo' | 'inativo';
  imagem: string;
  created_at: string;
  updated_at: string;
}

export function usePratos() {
  const [pratos, setPratos] = useState<Prato[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const fetchPratos = useCallback(async () => {
    try {
      setLoading(true);
      const { data, error } = await supabase
        .from('pratos')
        .select('*')
        .order('nome', { ascending: true });

      if (error) throw error;
      setPratos(data || []);
      setError(null);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Erro ao carregar pratos');
    } finally {
      setLoading(false);
    }
  }, []);

  const createPrato = useCallback(async (prato: Omit<Prato, 'id' | 'created_at' | 'updated_at'>) => {
    try {
      setLoading(true);
      const { data, error } = await supabase
        .from('pratos')
        .insert([prato])
        .select()
        .single();

      if (error) throw error;
      await fetchPratos();
      return data;
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Erro ao criar prato');
      return null;
    } finally {
      setLoading(false);
    }
  }, [fetchPratos]);

  const updatePrato = useCallback(async (id: string, updates: Partial<Prato>) => {
    try {
      setLoading(true);
      const { data, error } = await supabase
        .from('pratos')
        .update(updates)
        .eq('id', id)
        .select()
        .single();

      if (error) throw error;
      await fetchPratos();
      return data;
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Erro ao atualizar prato');
      return null;
    } finally {
      setLoading(false);
    }
  }, [fetchPratos]);

  const deletePrato = useCallback(async (id: string) => {
    try {
      setLoading(true);
      const { error } = await supabase
        .from('pratos')
        .delete()
        .eq('id', id);

      if (error) throw error;
      await fetchPratos();
      return true;
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Erro ao deletar prato');
      return false;
    } finally {
      setLoading(false);
    }
  }, [fetchPratos]);

  const getPratosAtivos = useCallback(() => {
    return pratos.filter(prato => prato.status === 'ativo');
  }, [pratos]);

  const getPratosPorCategoria = useCallback((categoria: string) => {
    return pratos.filter(prato => prato.categoria === categoria);
  }, [pratos]);

  useEffect(() => {
    fetchPratos();
  }, [fetchPratos]);

  return {
    pratos,
    loading,
    error,
    fetchPratos,
    createPrato,
    updatePrato,
    deletePrato,
    getPratosAtivos,
    getPratosPorCategoria,
  };
}
