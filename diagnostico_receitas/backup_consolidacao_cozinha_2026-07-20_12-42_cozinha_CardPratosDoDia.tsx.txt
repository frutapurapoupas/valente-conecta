'use client';

// ============================================
// COMPONENTE - CARD PRATOS DO DIA
// ============================================

interface Prato {
  id: string;
  nome: string;
  imagem_url?: string;
  preco: number;
  categoria: string;
}

import Link from 'next/link';
import { Utensils, Star } from 'lucide-react';

export function CardPratosDoDia({ pratos }: { pratos: Prato[] }) {
  if (!pratos || pratos.length === 0) {
    return (
      <div className="bg-gray-100 rounded-xl border border-gray-200 p-6 text-center text-gray-500">
        Nenhum prato do dia definido.
      </div>
    );
  }

  return (
    <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
      {pratos.slice(0, 3).map((prato) => (
        <div key={prato.id} className="bg-white rounded-xl border border-gray-200 p-4 hover:border-blue-300 hover:shadow-sm transition">
          <div className="flex items-start gap-4">
            <img
              src={prato.imagem_url || `https://images.unsplash.com/photo-1504674900247-0877df9cc836?w=100&h=100&fit=crop`}
              alt={prato.nome}
              className="w-20 h-20 rounded-lg object-cover flex-shrink-0"
            />
            <div className="flex-1">
              <h3 className="font-semibold text-gray-900">{prato.nome}</h3>
              <p className="text-sm text-gray-500">{prato.categoria}</p>
              <p className="text-lg font-bold text-green-400 mt-1">R$ {prato.preco.toFixed(2)}</p>
            </div>
          </div>
        </div>
      ))}
    </div>
  );
}

export default CardPratosDoDia;
