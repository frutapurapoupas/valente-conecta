import { supabase } from '@/lib/supabase';
import { LancamentoFinanceiroCozinha } from '../types/financeiro';

export class FinanceiroCozinhaRepository {
  async buscarLancamentos(
    tipo?: 'receita' | 'despesa',
    dataInicio?: string,
    dataFim?: string
  ): Promise<LancamentoFinanceiroCozinha[]> {
    let query = supabase
      .from('cozinha_financeiro_lancamentos')
      .select('*')
      .order('data', { ascending: false });

    if (tipo) {
      query = query.eq('tipo', tipo);
    }

    if (dataInicio) {
      query = query.gte('data', dataInicio);
    }

    if (dataFim) {
      query = query.lte('data', dataFim);
    }

    const { data, error } = await query;

    if (error) {
      console.error('Erro ao buscar lanÃ§amentos:', error);
      throw new Error('Falha ao carregar lanÃ§amentos');
    }

    return data || [];
  }

  async criarLancamento(
    lancamento: Omit<LancamentoFinanceiroCozinha, 'id' | 'created_at'>
  ): Promise<LancamentoFinanceiroCozinha> {
    const { data, error } = await supabase
      .from('cozinha_financeiro_lancamentos')
      .insert({
        ...lancamento,
        created_at: new Date().toISOString()
      })
      .select()
      .single();

    if (error) {
      console.error('Erro ao criar lanÃ§amento:', error);
      throw new Error('Falha ao criar lanÃ§amento');
    }

    return data;
  }
}
