// app/cozinha/services/receitaService.ts
// ⚠️ SERVIÇO - APENAS COMUNICAÇÃO COM API!

import { Receita } from '../hooks/useReceitas'

// Simulação de API - Substituir por chamada real
const API_URL = '/api/cozinha/receitas'

export const receitaService = {
  listar: async (): Promise<Receita[]> => {
    const response = await fetch(API_URL)
    if (!response.ok) throw new Error('Erro ao listar receitas')
    return response.json()
  },

  buscar: async (id: string): Promise<Receita> => {
    const response = await fetch(`${API_URL}/${id}`)
    if (!response.ok) throw new Error('Erro ao buscar receita')
    return response.json()
  },

  criar: async (dados: Omit<Receita, 'id'>): Promise<Receita> => {
    const response = await fetch(API_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(dados)
    })
    if (!response.ok) throw new Error('Erro ao criar receita')
    return response.json()
  },

  editar: async (id: string, dados: Partial<Receita>): Promise<Receita> => {
    const response = await fetch(`${API_URL}/${id}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(dados)
    })
    if (!response.ok) throw new Error('Erro ao editar receita')
    return response.json()
  },

  excluir: async (id: string): Promise<void> => {
    const response = await fetch(`${API_URL}/${id}`, {
      method: 'DELETE'
    })
    if (!response.ok) throw new Error('Erro ao excluir receita')
  }
}
