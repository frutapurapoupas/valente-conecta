interface Ingredient {
  id: string;
  name: string;
  unit: string;
  currentPrice: number;
  stock: number;
}

interface RecipeItem {
  ingredientId: string;
  name: string;
  quantity: number;
  unit: string;
  cost: number;
}

export const calcularCustoItem = (
  ingredient: Ingredient | undefined,
  quantity: number,
  unit: string
): { cost: number; falta: number } => {
  if (!ingredient) return { cost: 0, falta: 0 };
  
  const neededQuantity = quantity;
  const currentStock = ingredient.stock || 0;
  const falta = Math.max(0, neededQuantity - currentStock);
  
  let totalCost = 0;
  
  if (ingredient.unit === 'kg') {
    totalCost = (falta / 1000) * ingredient.currentPrice;
  } else if (ingredient.unit === 'g') {
    totalCost = (falta / 1000) * ingredient.currentPrice;
  } else if (ingredient.unit === 'L') {
    totalCost = (falta / 1000) * ingredient.currentPrice;
  } else if (ingredient.unit === 'mL') {
    totalCost = (falta / 1000) * ingredient.currentPrice;
  } else {
    totalCost = falta * ingredient.currentPrice;
  }
  
  return { cost: totalCost, falta };
};

export const calcularCustoTotal = (items: RecipeItem[]): number => {
  return items.reduce((sum, item) => sum + (item.cost || 0), 0);
};
