import { useState, useEffect, useCallback } from 'react';
import { ProdutoService } from '@/modules/cozinha/services/ProdutoService';
import type { Produto } from '@/modules/cozinha/types/produto';

export function useProdutos() {
  const [produtos, setProdutos] = useState<Produto[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  const carregarProdutos = useCallback(async () => {
    try {
      setLoading(true);
      setError(null);
      const data = await ProdutoService.listarProdutos();
      setProdutos(data);
    } catch (err) {
      setError(err instanceof Error ? err : new Error('Erro ao carregar produtos'));
      console.error('Erro ao carregar produtos:', err);
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    carregarProdutos();
  }, [carregarProdutos]);

  return {
    produtos,
    loading,
    error,
    carregarProdutos,
  };
}
