import { FinanceiroCozinhaRepository } from '../repositories/FinanceiroCozinhaRepository';
import { LancamentoFinanceiroCozinha, ResumoFinanceiroCozinha } from '../types/financeiro';

export class FinanceiroCozinhaService {
  private repository = new FinanceiroCozinhaRepository();

  async listarLancamentos(
    tipo?: 'receita' | 'despesa',
    dataInicio?: string,
    dataFim?: string
  ): Promise<LancamentoFinanceiroCozinha[]> {
    try {
      return await this.repository.buscarLancamentos(tipo, dataInicio, dataFim);
    } catch (error) {
      console.error('Erro no Service ao listar lanÃ§amentos:', error);
      throw error;
    }
  }

  async criarLancamento(lancamento: Omit<LancamentoFinanceiroCozinha, 'id' | 'created_at'>): Promise<LancamentoFinanceiroCozinha> {
    try {
      this.validarLancamento(lancamento);
      return await this.repository.criarLancamento(lancamento);
    } catch (error) {
      console.error('Erro no Service ao criar lanÃ§amento:', error);
      throw error;
    }
  }

  async buscarResumo(dataInicio?: string, dataFim?: string): Promise<ResumoFinanceiroCozinha> {
    try {
      const lancamentos = await this.repository.buscarLancamentos(undefined, dataInicio, dataFim);
      
      const receitas = lancamentos.filter(l => l.tipo === 'receita');
      const despesas = lancamentos.filter(l => l.tipo === 'despesa');

      const totalReceitas = receitas.reduce((sum, l) => sum + l.valor, 0);
      const totalDespesas = despesas.reduce((sum, l) => sum + l.valor, 0);

      const receitasPorCategoria: Record<string, number> = {};
      receitas.forEach(l => {
        receitasPorCategoria[l.categoria] = (receitasPorCategoria[l.categoria] || 0) + l.valor;
      });

      const despesasPorCategoria: Record<string, number> = {};
      despesas.forEach(l => {
        despesasPorCategoria[l.categoria] = (despesasPorCategoria[l.categoria] || 0) + l.valor;
      });

      return {
        totalReceitas,
        totalDespesas,
        saldo: totalReceitas - totalDespesas,
        receitasPorCategoria,
        despesasPorCategoria,
        ultimosLancamentos: lancamentos.slice(0, 10)
      };
    } catch (error) {
      console.error('Erro no Service ao buscar resumo:', error);
      throw error;
    }
  }

  private validarLancamento(lancamento: Omit<LancamentoFinanceiroCozinha, 'id' | 'created_at'>): void {
    if (!lancamento.tipo) throw new Error('Tipo de lanÃ§amento Ã© obrigatÃ³rio');
    if (!lancamento.categoria) throw new Error('Categoria Ã© obrigatÃ³ria');
    if (!lancamento.descricao || lancamento.descricao.trim().length < 3) {
      throw new Error('DescriÃ§Ã£o deve ter pelo menos 3 caracteres');
    }
    if (lancamento.valor <= 0) throw new Error('Valor deve ser maior que zero');
    if (!lancamento.data) throw new Error('Data Ã© obrigatÃ³ria');
  }
}
