"use client";

import { CardapioItem } from '../types/cardapio';

interface CardapioListProps {
  items: CardapioItem[];
  onRemove?: (id: string) => void;
  loading?: boolean;
}

const diasSemana = {
  segunda: 'Segunda-feira',
  terca: 'TerÃ§a-feira',
  quarta: 'Quarta-feira',
  quinta: 'Quinta-feira',
  sexta: 'Sexta-feira',
  sabado: 'SÃ¡bado',
  domingo: 'Domingo'
};

export function CardapioList({ items, onRemove, loading = false }: CardapioListProps) {
  if (loading) {
    return (
      <div className="flex justify-center items-center py-8">
        <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-500"></div>
      </div>
    );
  }

  if (items.length === 0) {
    return (
      <div className="text-center py-8">
        <p className="text-gray-500">Nenhum item no cardÃ¡pio</p>
      </div>
    );
  }

  const itemsPorDia = items.reduce((acc, item) => {
    const dia = item.dia;
    if (!acc[dia]) acc[dia] = [];
    acc[dia].push(item);
    return acc;
  }, {} as Record<string, CardapioItem[]>);

  return (
    <div className="space-y-6">
      {Object.entries(itemsPorDia).map(([dia, itens]) => (
        <div key={dia}>
          <h3 className="text-lg font-semibold text-gray-800 mb-3">
            {diasSemana[dia as keyof typeof diasSemana] || dia}
          </h3>
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-3">
            {itens.map((item) => (
              <div key={item.id} className="bg-gray-50 rounded-lg p-3 flex justify-between items-center">
                <div>
                  <p className="font-medium text-gray-800">
                    {item.produto?.nome || 'Produto nÃ£o encontrado'}
                  </p>
                  {item.precoEspecial && (
                    <p className="text-sm text-green-600">
                      R$ {item.precoEspecial.toFixed(2)}
                    </p>
                  )}
                </div>
                {onRemove && (
                  <button
                    onClick={() => onRemove(item.id)}
                    className="text-red-500 hover:text-red-700 transition-colors"
                  >
                    âœ•
                  </button>
                )}
              </div>
            ))}
          </div>
        </div>
      ))}
    </div>
  );
}
