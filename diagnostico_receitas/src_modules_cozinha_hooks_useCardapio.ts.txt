// app/cozinha/hooks/useCardapio.ts
"use client";

import { useState, useEffect, useCallback, useMemo } from 'react';
import { CardapioService } from '../services/CardapioService';
import { CardapioItem, CardapioInput } from '../types/cardapio';

export function useCardapio() {
  const [cardapio, setCardapio] = useState<CardapioItem[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const service = useMemo(() => new CardapioService(), []);

  const carregarCardapio = useCallback(async () => {
    try {
      setLoading(true);
      setError(null);
      const data = await service.listarTodos();
      setCardapio(data);
    } catch (err) {
      const message = err instanceof Error ? err.message : 'Erro ao carregar cardÃ¡pio';
      setError(message);
      console.error('useCardapio - Erro:', err);
    } finally {
      setLoading(false);
    }
  }, [service]);

  const buscarPorDia = useCallback(async (dia: string) => {
    try {
      setLoading(true);
      setError(null);
      const data = await service.listarPorDia(dia);
      return data;
    } catch (err) {
      const message = err instanceof Error ? err.message : 'Erro ao buscar cardÃ¡pio do dia';
      setError(message);
      console.error('useCardapio - Erro ao buscar por dia:', err);
      throw err;
    } finally {
      setLoading(false);
    }
  }, [service]);

  const adicionarItem = useCallback(async (item: CardapioInput) => {
    try {
      setLoading(true);
      setError(null);
      const novo = await service.adicionar(item);
      setCardapio(prev => [...prev, novo]);
      return novo;
    } catch (err) {
      const message = err instanceof Error ? err.message : 'Erro ao adicionar item ao cardÃ¡pio';
      setError(message);
      console.error('useCardapio - Erro ao adicionar:', err);
      throw err;
    } finally {
      setLoading(false);
    }
  }, [service]);

  const removerItem = useCallback(async (id: string) => {
    try {
      setLoading(true);
      setError(null);
      await service.remover(id);
      setCardapio(prev => prev.filter(item => item.id !== id));
    } catch (err) {
      const message = err instanceof Error ? err.message : 'Erro ao remover item do cardÃ¡pio';
      setError(message);
      console.error('useCardapio - Erro ao remover:', err);
      throw err;
    } finally {
      setLoading(false);
    }
  }, [service]);

  const atualizarItem = useCallback(async (id: string, dados: Partial<CardapioInput>) => {
    try {
      setLoading(true);
      setError(null);
      const atualizado = await service.atualizar(id, dados);
      setCardapio(prev => prev.map(item => item.id === id ? atualizado : item));
      return atualizado;
    } catch (err) {
      const message = err instanceof Error ? err.message : 'Erro ao atualizar item do cardÃ¡pio';
      setError(message);
      console.error('useCardapio - Erro ao atualizar:', err);
      throw err;
    } finally {
      setLoading(false);
    }
  }, [service]);

  useEffect(() => {
    carregarCardapio();
  }, [carregarCardapio]);

  return {
    cardapio,
    loading,
    error,
    carregarCardapio,
    buscarPorDia,
    adicionarItem,
    removerItem,
    atualizarItem
  };
}
