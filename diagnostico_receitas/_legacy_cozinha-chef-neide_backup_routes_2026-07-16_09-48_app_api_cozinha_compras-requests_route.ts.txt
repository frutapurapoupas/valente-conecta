import { NextRequest, NextResponse } from 'next/server';

const requestsPath = path.join(process.cwd(), 'data', 'compras_requests.json');

function ensureFile(filePath: string) {
  if (!fs.existsSync(filePath)) {
    fs.writeFileSync(filePath, JSON.stringify([], null, 2));
  }
}

function readData(filePath: string) {
  ensureFile(filePath);
  return JSON.parse(fs.readFileSync(filePath, 'utf-8'));
}

function writeData(filePath: string, data: any) {
  fs.writeFileSync(filePath, JSON.stringify(data, null, 2));
}

function normalizeUnit(unit: string) {
  const u = String(unit || 'un').trim().toLowerCase();
  if (u === 'grama' || u === 'gramas') return 'g';
  if (u === 'mililitro' || u === 'mililitros') return 'ml';
  if (u === 'litro' || u === 'litros') return 'L';
  if (u === 'quilo' || u === 'quilos') return 'kg';
  if (u === 'unidade') return 'un';
  return u;
}

function toPurchaseUnit(quantity: number, unit: string) {
  const normalized = normalizeUnit(unit);
  if (normalized === 'g') {
    return { quantity: quantity / 1000, unit: 'kg' };
  }
  if (normalized === 'ml') {
    return { quantity: quantity / 1000, unit: 'L' };
  }
  return { quantity, unit: normalized };
}

function upsertCompraUnica(compras: any[], ingrediente: any, origem: string) {
  const keyNome = String(ingrediente.ingredientName || ingrediente.nome || '').trim().toLowerCase();
  const converted = toPurchaseUnit(
    Number(ingrediente.quantidade || ingrediente.quantity || 0),
    String(ingrediente.unit || ingrediente.unidade || 'un')
  );
  const keyUnidade = normalizeUnit(converted.unit);

  const existing = compras.find((item) =>
    String(item.nome || '').trim().toLowerCase() === keyNome &&
    String(item.unidade || '').trim().toLowerCase() === keyUnidade
  );

  const quantidade = Number(converted.quantity || 0);
  const precoEstimado = Number(ingrediente.preco_estimado || ingrediente.price || 0);

  if (existing) {
    existing.quantidade = Number(existing.quantidade || 0) + quantidade;
    if (!existing.preco_estimado || existing.preco_estimado === 0) {
      existing.preco_estimado = precoEstimado;
    }
    existing.receita_origem = existing.receita_origem
      ? `${existing.receita_origem}; ${origem}`
      : origem;
    existing.updated_at = new Date().toISOString();
    return;
  }

  compras.push({
    id: Date.now().toString() + Math.floor(Math.random() * 999).toString(),
    nome: ingrediente.ingredientName || ingrediente.nome,
    unidade: keyUnidade,
    quantidade,
    preco_estimado: precoEstimado,
    preco_real: 0,
    fornecedor: '',
    comprado: false,
    prioridade: 'media',
    receita_origem: origem,
    observacao: '',
    created_at: new Date().toISOString(),
    updated_at: new Date().toISOString()
  });
}

export async function POST(request: NextRequest) {
  try {
    const body = await request.json();
    const items = readData(requestsPath);

    const newItem = {
      id: `${Date.now()}-${Math.floor(Math.random() * 1000000)}`,
      receitaId: body.receitaId,
      receitaNome: body.receitaNome,
      quantidadeProduzir: Number(body.quantidadeProduzir || 1),
      ingredientes: Array.isArray(body.ingredientes) ? body.ingredientes : [],
      status: 'pendente',
      createdAt: new Date().toISOString(),
      approvedAt: null
